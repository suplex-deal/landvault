1pub fn check_slippage(expected_out: u128, actual_out: u128, max_bps: u32) -> Result<(), SlippageError> {
2    if expected_out == 0 { return Err(SlippageError::ZeroExpected); }
3    let loss = expected_out.saturating_mul(max_bps as u128).saturating_div(10_000);
4    if actual_out < expected_out.saturating_sub(loss) {
5        return Err(SlippageError::Exceeded);
6    }
7    Ok(())
8}
