1//! #1109: validator staking and slashing.
2//!
3//! `ValidatorStaking` is stored on the contract
4//! ([`crate::PropertyBridge::staking`]) and tracks the credit-based stake of
5//! each registered validator. Approvals for bridge requests require a
6//! **staked-weight quorum** (`STAKED_QUORUM_BPS` of `total_staked`), not a
7//! bare signature count; a validator that flip-flops its vote on a request is
8//! slashed `SLASH_PERCENT` of its current stake into the slash pool.
9//!
10//! The stake is a nominal economic bond held in the ledger (no token
11//! transfers): validators self-stake in `PropertyBridge::stake_validator`,
12//! withdraw via `withdraw_stake`, and are penalised via `slash_validator` or
13//! automatically on a conflicting vote.
14
15use ink::primitives::AccountId;
16use ink::storage::Mapping;
17
18/// Minimum stake required for a validator to become economically active.
19pub const MIN_VALIDATOR_STAKE: u128 = 10_000_000;
20/// Percentage of a validator's stake removed on a slash.
21pub const SLASH_PERCENT: u128 = 20;
22/// Percentage of `total_staked` required to lock/approve a bridge request.
23pub const STAKED_QUORUM_BPS: u128 = 6_000;
24
25/// Errors returned by the staking ledger.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum StakeError {
28    /// The requested stake amount is below [`MIN_VALIDATOR_STAKE`].
29    BelowMinimum,
30    /// The account does not hold enough stake for the requested withdrawal.
31    InsufficientStake,
32}
33
34/// Credit-based validator staking ledger.
35#[ink::storage_item]
36#[derive(Default)]
37pub struct ValidatorStaking {
38    /// Current stake held by each account.
39    pub stakes: Mapping<AccountId, u128>,
40    /// Sum of all stakes; the denominator for the quorum computation.
41    pub total_staked: u128,
42    /// Accumulated slashed stake, reserved for future redemption or burns.
43    pub slash_pool: u128,
44}
45
46impl core::fmt::Debug for ValidatorStaking {
47    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
48        f.debug_struct("ValidatorStaking")
49            .field("total_staked", &self.total_staked)
50            .field("slash_pool", &self.slash_pool)
51            .finish()
52    }
53}
54
55impl ValidatorStaking {
56    /// Credited `amount` stake to `account`.
57    pub fn stake(&mut self, account: AccountId, amount: u128) -> Result<(), StakeError> {
58        if amount < MIN_VALIDATOR_STAKE {
59            return Err(StakeError::BelowMinimum);
60        }
61        let current = self.stakes.get(account).unwrap_or(0);
62        self.stakes.insert(account, &(current + amount));
63        self.total_staked = self.total_staked.saturating_add(amount);
64        Ok(())
65    }
66
67    /// Withdraws `amount` from `account`'s stake.
68    pub fn unstake(&mut self, account: AccountId, amount: u128) -> Result<u128, StakeError> {
69        let current = self.stakes.get(account).unwrap_or(0);
70        if amount == 0 || amount > current {
71            return Err(StakeError::InsufficientStake);
72        }
73        self.stakes.insert(account, &(current - amount));
74        self.total_staked = self.total_staked.saturating_sub(amount);
75        Ok(amount)
76    }
77
78    /// Slashes `SLASH_PERCENT` of `account`'s stake into the slash pool.
79    /// Returns the slashed amount (0 when the account holds no stake).
80    pub fn slash(&mut self, account: AccountId) -> u128 {
81        let current = self.stakes.get(account).unwrap_or(0);
82        if current == 0 {
83            return 0;
84        }
85        let penalty = current * SLASH_PERCENT / 100;
86        self.stakes.insert(account, &(current - penalty));
87        self.total_staked = self.total_staked.saturating_sub(penalty);
88        self.slash_pool = self.slash_pool.saturating_add(penalty);
89        penalty
90    }
91
92    /// Whether `account` currently holds the active-stake threshold.
93    pub fn is_active(&self, account: AccountId) -> bool {
94        self.stakes.get(account).unwrap_or(0) >= MIN_VALIDATOR_STAKE
95    }
96
97    /// The staked weight of `account` (0 for non-stakers).
98    pub fn weight(&self, account: AccountId) -> u128 {
99        self.stakes.get(account).unwrap_or(0)
100    }
101
102    /// Total staked across all validators.
103    pub fn total(&self) -> u128 {
104        self.total_staked
105    }
106
107    /// The slash pool accumulated so far.
108    pub fn pool(&self) -> u128 {
109        self.slash_pool
110    }
111}

113#[cfg(test)]
114mod tests {
115    use super::*;
116
117    #[ink::test]
118    fn stake_enforces_minimum() {
119        let mut ledger = ValidatorStaking::default();
120        let alice = AccountId::from([1u8; 32]);
121        assert_eq!(
122            ledger.stake(alice, MIN_VALIDATOR_STAKE - 1),
123            Err(StakeError::BelowMinimum)
124        );
125        assert_eq!(ledger.stake(alice, MIN_VALIDATOR_STAKE), Ok(()));
126        assert!(ledger.is_active(alice));
127        assert_eq!(ledger.weight(alice), MIN_VALIDATOR_STAKE);
128        assert_eq!(ledger.total(), MIN_VALIDATOR_STAKE);
129    }
130
131    #[ink::test]
132    fn unstake_rejects_insufficient() {
133        let mut ledger = ValidatorStaking::default();
134        let alice = AccountId::from([1u8; 32]);
135        ledger.stake(alice, MIN_VALIDATOR_STAKE).unwrap();
136        assert_eq!(
137            ledger.unstake(alice, MIN_VALIDATOR_STAKE + 1),
138            Err(StakeError::InsufficientStake)
139        );
140        assert_eq!(
141            ledger.unstake(alice, MIN_VALIDATOR_STAKE),
142            Ok(MIN_VALIDATOR_STAKE)
143        );
144        assert_eq!(ledger.total(), 0);
145        assert!(!ledger.is_active(alice));
146    }
147
148    #[ink::test]
149    fn slash_removes_percent_into_pool() {
150        let mut ledger = ValidatorStaking::default();
151        let alice = AccountId::from([1u8; 32]);
152        ledger.stake(alice, 100_000_000).unwrap();
153        let penalty = ledger.slash(alice);
154        assert_eq!(penalty, 20_000_000);
155        assert_eq!(ledger.weight(alice), 80_000_000);
156        assert_eq!(ledger.pool(), 20_000_000);
157        assert_eq!(ledger.total(), 80_000_000);
158        // Slashing a non-staker is a no-op.
159        assert_eq!(ledger.slash(AccountId::from([2u8; 32])), 0);
160    }
161
162    #[ink::test]
163    fn quorum_bps_is_percentage_of_total() {
164        let mut ledger = ValidatorStaking::default();
165        let alice = AccountId::from([1u8; 32]);
166        let bob = AccountId::from([2u8; 32]);
167        ledger.stake(alice, 100_000_000).unwrap();
168        ledger.stake(bob, 100_000_000).unwrap();
169        let quorum = ledger.total() * STAKED_QUORUM_BPS / 10_000;
170        assert_eq!(quorum, 120_000_000);
171    }
172}
