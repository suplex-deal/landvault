1//! Closes #803: public `compute_borrow_rate` view so off-chain tools can
2//! quote without simulating contract state. Wired into `landvault_lending`
3//! (#1091): the `borrow_rate` message and the pool-level interest accrual in
4//! `deposit`/`borrow` both use this linear model, documented in the message
5//! docs as `rate_bps = 200 + 1000 × utilisation_bps / 10 000`.
6
7const BASE_RATE_BPS: u32 = 200; // 2%
8const SLOPE_BPS: u32 = 1_000; // +10% at 100% utilisation
9
10/// Computes the borrow rate (in basis points) for a given utilisation,
11/// expressed in basis points (0-10_000).
12pub fn compute_borrow_rate(utilisation_bps: u32) -> u32 {
13    let utilisation_bps = utilisation_bps.min(10_000);
14    BASE_RATE_BPS + (SLOPE_BPS as u64 * utilisation_bps as u64 / 10_000) as u32
15}
16
17#[cfg(test)]
18mod tests {
19    use super::*;
20
21    #[test]
22    fn zero_utilisation_yields_base_rate() {
23        assert_eq!(compute_borrow_rate(0), BASE_RATE_BPS);
24    }
25
26    #[test]
27    fn full_utilisation_yields_base_plus_slope() {
28        assert_eq!(compute_borrow_rate(10_000), BASE_RATE_BPS + SLOPE_BPS);
29    }
30
31    #[test]
32    fn utilisation_above_100_percent_is_clamped() {
33        assert_eq!(compute_borrow_rate(50_000), compute_borrow_rate(10_000));
34    }
35}
