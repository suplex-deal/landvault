15/// Dynamic premium calculation with comprehensive risk factors.
16///
17/// Steps:
18///  1. Base rate (actuarial model or coverage-type default)
19///  2. Risk multiplier (weighted location / construction / age / claims-history scores)
20///  3. Coverage-type multiplier
21///  4. Pool-utilization multiplier
22///  5. Time-based multiplier
23///  6. Discount multiplier (multi-policy, claim-free, safety, loyalty)
24///  7. **Claim-frequency multiplier** (rolling-window surcharge — new)
25///
26/// The claim-frequency step uses `modifiers.recent_claims_count`, which the
27/// caller populates with the number of approved claims filed against the
28/// property within the rolling observation window (default: 12 months).
29/// Defaults match the table in `DYNAMIC_PREMIUM_CALCULATION.md`.
30pub fn calculate_dynamic_premium(
31    risk_assessment: &RiskAssessment,
32    coverage_amount: u128,
33    coverage_type: &CoverageType,
34    pool: &RiskPool,
35    actuarial_model: Option<&ActuarialModel>,
36    modifiers: &PremiumModifiers,
37    policy_duration_seconds: u64,
38) -> PremiumCalculation {
39    // Step 1: Calculate base rate from actuarial model or use default
40    let base_rate = calculate_base_rate(actuarial_model, coverage_type);
41
42    // Step 2: Calculate comprehensive risk multiplier
43    let risk_multiplier = calculate_risk_multiplier(risk_assessment);
44
45    // Step 3: Calculate coverage type multiplier
46    let coverage_multiplier = coverage_type_multiplier(coverage_type);
47
48    // Step 4: Calculate pool utilization adjustment
49    let pool_utilization_multiplier = calculate_pool_utilization_multiplier(pool);
50
51    // Step 5: Calculate time-based adjustments
52    let time_multiplier = calculate_time_multiplier(policy_duration_seconds);
53
54    // Step 6: Calculate discounts
55    let discount_multiplier = calculate_discount_multiplier(modifiers);
56
57    // Step 7: Rolling-window claim-frequency surcharge
58    // recent_claims_count = number of approved claims in the observation window
59    let claim_freq_multiplier =
60        calculate_claim_frequency_multiplier(modifiers.recent_claims_count);
61
62    // Formula: coverage × base_rate × risk × coverage × pool × time × discount × claim_freq
63    // All multipliers are in basis points (10 000 = 1.0×), divisor accounts for all 7 factors.
64    let annual_premium = coverage_amount
65        .saturating_mul(base_rate as u128)
66        .saturating_mul(risk_multiplier as u128)
67        .saturating_mul(coverage_multiplier as u128)
68        .saturating_mul(pool_utilization_multiplier as u128)
69        .saturating_mul(time_multiplier as u128)
70        .saturating_mul(discount_multiplier as u128)
71        .saturating_mul(claim_freq_multiplier as u128)
72        / PREMIUM_CALCULATION_DIVISOR;
73
74    // Prorate for policy duration
75    let duration_premium =
76        annual_premium.saturating_mul(policy_duration_seconds as u128) / SECONDS_PER_YEAR;
77
78    let monthly_premium = duration_premium / 12;
79
80    // Calculate dynamic deductible based on risk
81    let deductible = calculate_deductible(coverage_amount, risk_assessment, modifiers);
82
83    PremiumCalculation {
84        base_rate,
85        risk_multiplier,
86        coverage_multiplier,
87        pool_utilization_multiplier,
88        time_multiplier,
89        discount_multiplier,
90        claim_freq_multiplier,
91        annual_premium: duration_premium,
92        monthly_premium,
93        deductible,
94        breakdown: PremiumBreakdown {
95            base_premium: coverage_amount.saturating_mul(base_rate as u128)
96                / BASIS_POINTS_DENOMINATOR,
97            risk_adjustment: calculate_risk_adjustment_amount(
98                coverage_amount,
99                base_rate,
100                risk_multiplier,
101            ),
102            coverage_adjustment: calculate_coverage_adjustment_amount(
103                coverage_amount,
104                base_rate,
105                risk_multiplier,
106                coverage_multiplier,
107            ),
108            pool_adjustment: calculate_pool_adjustment_amount(
109                coverage_amount,
110                base_rate,
111                risk_multiplier,
112                coverage_multiplier,
113                pool_utilization_multiplier,
114            ),
115            time_adjustment: calculate_time_adjustment_amount(
116                coverage_amount,
117                base_rate,
118                risk_multiplier,
119                coverage_multiplier,
120                pool_utilization_multiplier,
121                time_multiplier,
122            ),
123            discount_amount: calculate_discount_amount(
124                coverage_amount,
125                base_rate,
126                risk_multiplier,
127                coverage_multiplier,
128                pool_utilization_multiplier,
129                time_multiplier,
130                discount_multiplier,
131            ),
132            claim_freq_adjustment: calculate_claim_freq_adjustment_amount(
133                coverage_amount,
134                base_rate,
135                risk_multiplier,
136                coverage_multiplier,
137                pool_utilization_multiplier,
138                time_multiplier,
139                discount_multiplier,
140                claim_freq_multiplier,
141            ),
142        },
143    }
144}

146/// Calculate base rate from actuarial model
147fn calculate_base_rate(
148    actuarial_model: Option<&ActuarialModel>,
149    coverage_type: &CoverageType,
150) -> u32 {
151    match actuarial_model {
152        Some(model) => {
153            // Use actuarial model: expected_loss_ratio * confidence_adjustment
154            // Expected loss ratio in basis points (e.g., 600 = 6%)
155            let expected_loss = model.expected_loss_ratio;
156
157            // Confidence level adjustment (95% = 1.0, 99% = 1.2)
158            let confidence_adjustment = match model.confidence_level {
159                95 => 100,
160                96 => 105,
161                97 => 110,
162                98 => 115,
163                99 => 120,
164                _ => 100,
165            };
166
167            // Base rate = expected_loss * confidence_adjustment / 100
168            let model_rate = expected_loss.saturating_mul(confidence_adjustment as u32) / 100;
169
170            // Add expense loading (20% for operational costs)
171            model_rate.saturating_mul(120) / 100
172        }
173        None => {
174            // Default rates by coverage type (in basis points)
175            coverage_type_base_rate(coverage_type)
176        }
177    }
178}

180/// Default base rates by coverage type
181fn coverage_type_base_rate(coverage_type: &CoverageType) -> u32 {
182    match coverage_type {
183        CoverageType::Fire => 120,            // 1.2%
184        CoverageType::Flood => 200,           // 2.0%
185        CoverageType::Earthquake => 250,      // 2.5%
186        CoverageType::Theft => 100,           // 1.0%
187        CoverageType::LiabilityDamage => 150, // 1.5%
188        CoverageType::NaturalDisaster => 220, // 2.2%
189        CoverageType::Comprehensive => 300,   // 3.0%
190    }
191}

193/// Calculate comprehensive risk multiplier from assessment scores
194fn calculate_risk_multiplier(assessment: &RiskAssessment) -> u32 {
195    // Weighted average of risk components
196    // Location risk: 30%, Construction risk: 25%, Age risk: 20%, Claims history: 25%
197    let weighted_score = assessment
198        .location_risk_score
199        .saturating_mul(30)
200        .saturating_add(assessment.construction_risk_score.saturating_mul(25))
201        .saturating_add(assessment.age_risk_score.saturating_mul(20))
202        .saturating_add(assessment.claims_history_score.saturating_mul(25))
203        / 100;
204
205    // Convert score (0-100) to multiplier (50-400 basis points)
206    // Score 0 = very high risk (4.0x), Score 100 = very low risk (0.5x)
207    match weighted_score {
208        0..=10 => 400,  // Very high risk
209        11..=20 => 350, // High risk
210        21..=30 => 300, // High-medium risk
211        31..=40 => 250, // Medium-high risk
212        41..=50 => 200, // Medium risk
213        51..=60 => 170, // Medium-low risk
214        61..=70 => 140, // Low-medium risk
215        71..=80 => 110, // Low risk
216        81..=90 => 85,  // Very low risk
217        _ => 60,        // Minimal risk
218    }
219}

234/// Pool utilization adjustment
235/// Higher utilization = higher premiums to manage risk
236fn calculate_pool_utilization_multiplier(pool: &RiskPool) -> u32 {
237    if pool.total_capital == 0 {
238        return 200; // Default high multiplier if no capital
239    }
240
241    // Utilization rate: (total_capital - available_capital) / total_capital
242    let utilized = pool.total_capital.saturating_sub(pool.available_capital);
243    let utilization_rate = utilized.saturating_mul(100) / pool.total_capital;
244
245    // Adjust multiplier based on utilization
246    match utilization_rate {
247        0..=30 => 90,   // Low utilization - discount
248        31..=50 => 100, // Normal utilization
249        51..=70 => 115, // Medium-high utilization - slight increase
250        71..=85 => 135, // High utilization - significant increase
251        _ => 160,       // Critical utilization - major increase
252    }
253}

255/// Time-based adjustment
256/// Longer policies get slight discounts for stability
257fn calculate_time_multiplier(duration_seconds: u64) -> u32 {
258    match duration_seconds {
259        0..=2_592_000 => 105,          // < 30 days - short term premium
260        2_592_001..=7_776_000 => 100,  // 1-3 months - standard
261        7_776_001..=15_552_000 => 95,  // 3-6 months - slight discount
262        15_552_001..=31_536_000 => 90, // 6-12 months - good discount
263        _ => 85,                       // > 1 year - best discount
264    }
265}

267/// Rolling-window claim-frequency surcharge multiplier.
268///
269/// Uses the number of approved claims filed within the rolling observation
270/// window (typically 12 months) stored in `PremiumModifiers::recent_claims_count`.
271///
272/// | Claims in window | Multiplier | Basis points |
273/// |-----------------|------------|-------------|
274/// | 0               | 1.00×      | 10 000       | (no surcharge — baseline)
275/// | 1               | 1.10×      | 11 000       |
276/// | 2               | 1.25×      | 12 500       |
277/// | 3               | 1.50×      | 15 000       |
278/// | 4               | 1.75×      | 17 500       |
279/// | 5+              | 2.00×      | 20 000       | (high-frequency cap)
280fn calculate_claim_frequency_multiplier(recent_claims_count: u32) -> u32 {
281    match recent_claims_count {
282        0 => 10_000,  // 1.00× — no surcharge
283        1 => 11_000,  // 1.10×
284        2 => 12_500,  // 1.25×
285        3 => 15_000,  // 1.50×
286        4 => 17_500,  // 1.75×
287        _ => 20_000,  // 2.00× — high-frequency cap (5+ claims)
288    }
289}

291/// Calculate discount multiplier from modifiers
292fn calculate_discount_multiplier(modifiers: &PremiumModifiers) -> u32 {
293    let mut total_discount_bps: u32 = 0;
294
295    // Multi-policy discount (up to 15%)
296    if modifiers.has_multiple_policies {
297        total_discount_bps = total_discount_bps.saturating_add(1500);
298    }
299
300    // Claim-free discount (up to 20% based on years)
301    if modifiers.claim_free_years > 0 {
302        let claim_free_discount = match modifiers.claim_free_years {
303            1 => 500,  // 5%
304            2 => 1000, // 10%
305            3 => 1500, // 15%
306            _ => 2000, // 20% for 4+ years
307        };
308        total_discount_bps = total_discount_bps.saturating_add(claim_free_discount);
309    }
310
311    // Safety features discount (up to 10%)
312    if modifiers.has_safety_features {
313        total_discount_bps = total_discount_bps.saturating_add(1000);
314    }
315
316    // Loyalty discount (up to 10%)
317    if modifiers.loyalty_years > 0 {
318        let loyalty_discount = match modifiers.loyalty_years {
319            1..=2 => 300, // 3%
320            3..=5 => 600, // 6%
321            _ => 1000,    // 10% for 6+ years
322        };
323        total_discount_bps = total_discount_bps.saturating_add(loyalty_discount);
324    }
325
326    // Cap total discount at 40%
327    if total_discount_bps > 4000 {
328        total_discount_bps = 4000;
329    }
330
331    // Convert discount to multiplier (10000 - discount_bps)
332    10_000u32.saturating_sub(total_discount_bps)
333}

335/// Calculate deductible based on risk and modifiers
336fn calculate_deductible(
337    coverage_amount: u128,
338    assessment: &RiskAssessment,
339    modifiers: &PremiumModifiers,
340) -> u128 {
341    // Base deductible: 5% of coverage
342    let base_deductible_rate: u32 = 500; // 5% in basis points
343
344    // Adjust based on risk (higher risk = higher deductible)
345    let risk_adjustment: u32 = match assessment.overall_risk_score {
346        0..=20 => 200,  // Very high risk - 20% deductible
347        21..=40 => 150, // High risk - 15%
348        41..=60 => 100, // Medium risk - 10%
349        61..=80 => 75,  // Low risk - 7.5%
350        _ => 50,        // Very low risk - 5%
351    };
352
353    let deductible_rate = base_deductible_rate.saturating_add(risk_adjustment);
354
355    // Apply safety feature reduction
356    let reduction: u32 = 50;
357    let final_rate = if modifiers.has_safety_features {
358        deductible_rate.saturating_sub(reduction)
359    } else {
360        deductible_rate
361    };
362
363    coverage_amount.saturating_mul(final_rate as u128) / 10_000
364}

366/// Calculate risk adjustment amount for breakdown
367fn calculate_risk_adjustment_amount(coverage: u128, base_rate: u32, risk_multiplier: u32) -> u128 {
368    let base_premium = coverage.saturating_mul(base_rate as u128) / BASIS_POINTS_DENOMINATOR;
369    let risk_adjusted =
370        base_premium.saturating_mul(risk_multiplier as u128) / BASIS_POINTS_DENOMINATOR;
371    risk_adjusted.saturating_sub(base_premium)
372}

374/// Calculate coverage adjustment amount (difference that coverage_multiplier adds)
375fn calculate_coverage_adjustment_amount(
376    coverage: u128,
377    base_rate: u32,
378    risk_multiplier: u32,
379    coverage_multiplier: u32,
380) -> u128 {
381    let premium_before = coverage
382        .saturating_mul(base_rate as u128)
383        .saturating_mul(risk_multiplier as u128)
384        / PREMIUM_CALCULATION_DIVISOR;
385
386    let premium_after =
387        premium_before.saturating_mul(coverage_multiplier as u128) / BASIS_POINTS_DENOMINATOR;
388
389    premium_after.saturating_sub(premium_before)
390}

417    coverage_multiplier: u32,
418    pool_multiplier: u32,

434/// Calculate discount amount
435fn calculate_discount_amount(
436    coverage: u128,
437    base_rate: u32,
438    risk_multiplier: u32,
439    coverage_multiplier: u32,
440    pool_multiplier: u32,
441    time_multiplier: u32,
442    discount_multiplier: u32,
443) -> u128 {
444    let premium_before_discount = coverage
445        .saturating_mul(base_rate as u128)
446        .saturating_mul(risk_multiplier as u128)
447        .saturating_mul(coverage_multiplier as u128)
448        .saturating_mul(pool_multiplier as u128)
449        .saturating_mul(time_multiplier as u128)
450        / PREMIUM_CALCULATION_DIVISOR_5MULT;
451
452    let final_premium = premium_before_discount.saturating_mul(discount_multiplier as u128)
453        / BASIS_POINTS_DENOMINATOR;
454
455    premium_before_discount.saturating_sub(final_premium)
456}

458/// Calculate the claim-frequency surcharge amount for the breakdown.
459///
460/// This is the additional cost the claim-frequency multiplier adds on top of
461/// the already-discounted premium (i.e. it is always additive / non-negative).
462fn calculate_claim_freq_adjustment_amount(
463    coverage: u128,
464    base_rate: u32,
465    risk_multiplier: u32,
466    coverage_multiplier: u32,
467    pool_multiplier: u32,
468    time_multiplier: u32,
469    discount_multiplier: u32,
470    claim_freq_multiplier: u32,
471) -> u128 {
472    // Premium after discount, before claim-frequency step
473    let pre_freq = coverage
474        .saturating_mul(base_rate as u128)
475        .saturating_mul(risk_multiplier as u128)
476        .saturating_mul(coverage_multiplier as u128)
477        .saturating_mul(pool_multiplier as u128)
478        .saturating_mul(time_multiplier as u128)
479        .saturating_mul(discount_multiplier as u128)
480        / PREMIUM_CALCULATION_DIVISOR;
481
482    let post_freq = pre_freq
483        .saturating_mul(claim_freq_multiplier as u128)
484        / BASIS_POINTS_DENOMINATOR;
485
486    post_freq.saturating_sub(pre_freq)
487}

489// Constants
490const BASIS_POINTS_DENOMINATOR: u128 = 10_000;
491const SECONDS_PER_YEAR: u128 = 31_536_000; // 365 * 24 * 60 * 60
492/// Divisor for the 7-factor chain: base_rate × risk × coverage × pool × time × discount × claim_freq
493/// All 7 multipliers are in basis points (10 000 = 1.0×), so each divides by 10 000.
494/// 10 000^6 = 10^24; but base_rate uses a different scale (it's in raw basis points ÷ 10 000 once),
495/// so the combined divisor keeps the same structure as before with an extra ×10 000 for claim_freq.
496const PREMIUM_CALCULATION_DIVISOR: u128 = 10_000_000_000_000_000_000_000_000; // 10^25 for 7 factors
497const PREMIUM_CALCULATION_DIVISOR_LARGE: u128 = 10_000_000_000_000_000_000_000; // 10^22 for 6 multipliers (breakdown helpers)
498const PREMIUM_CALCULATION_DIVISOR_5MULT: u128 = 1_000_000_000_000_000_000_000; // 10^21 for discount calc (5 factors before discount)
